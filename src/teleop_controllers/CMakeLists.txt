cmake_minimum_required(VERSION 3.8)
project(teleop_controllers)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
find_package(ament_cmake REQUIRED)
find_package(Franka REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(safety_filter REQUIRED)
find_package(franka_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(quest_teleop_interfaces REQUIRED)
add_library(control_core src/control_core.cpp src/homing_core.cpp src/joint_motion_generator.cpp)
target_include_directories(control_core PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<INSTALL_INTERFACE:include>)
target_link_libraries(control_core PUBLIC Eigen3::Eigen safety_filter::cbf_safety_filter)
add_executable(cartesian_impedance_controller src/cartesian_impedance_controller.cpp)
ament_target_dependencies(cartesian_impedance_controller Franka franka_msgs geometry_msgs rclcpp sensor_msgs quest_teleop_interfaces)
target_link_libraries(cartesian_impedance_controller control_core)
install(TARGETS cartesian_impedance_controller DESTINATION lib/${PROJECT_NAME})
install(TARGETS control_core EXPORT export_control_core ARCHIVE DESTINATION lib LIBRARY DESTINATION lib)
install(DIRECTORY include/ DESTINATION include)
install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME})
install(FILES LICENSE DESTINATION share/${PROJECT_NAME})
install(DIRECTORY THIRD_PARTY_LICENSES DESTINATION share/${PROJECT_NAME})
ament_export_targets(export_control_core HAS_LIBRARY_TARGET)
ament_export_dependencies(Eigen3 safety_filter)
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_control_core test/test_control_core.cpp)
  target_link_libraries(test_control_core control_core)
  ament_add_gtest(test_homing_core test/test_homing_core.cpp)
  target_link_libraries(test_homing_core control_core)
  find_package(ament_cmake_pytest REQUIRED)
  ament_add_pytest_test(test_controller_mock test/test_controller_mock.py TIMEOUT 60)
  ament_add_pytest_test(test_homing_mock test/test_homing_mock.py TIMEOUT 60)
  ament_add_pytest_test(test_cbf_runtime test/test_cbf_runtime.py TIMEOUT 60)
endif()
ament_package()
