cmake_minimum_required(VERSION 3.8)
project(teleop_controllers)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
find_package(ament_cmake REQUIRED)
find_package(Franka REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(safety_filter REQUIRED)
find_package(franka_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(quest_teleop_interfaces REQUIRED)
add_library(controller_manager
  src/controller_manager.cpp src/cartesian_impedance_controller.cpp
  src/orientation_only_controller.cpp
  src/position_hold_controller.cpp src/damping_controller.cpp
  src/homing_controller.cpp src/joint_motion_generator.cpp)
target_include_directories(controller_manager PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<INSTALL_INTERFACE:include>)
target_link_libraries(controller_manager PUBLIC Eigen3::Eigen safety_filter::cbf_safety_filter)
add_executable(cartesian_impedance_controller src/controller_runtime.cpp)
ament_target_dependencies(cartesian_impedance_controller Franka franka_msgs geometry_msgs rclcpp sensor_msgs quest_teleop_interfaces)
target_link_libraries(cartesian_impedance_controller controller_manager)
install(TARGETS cartesian_impedance_controller DESTINATION lib/${PROJECT_NAME})
install(TARGETS controller_manager EXPORT export_controller_manager ARCHIVE DESTINATION lib LIBRARY DESTINATION lib)
install(DIRECTORY include/ DESTINATION include)
install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME})
install(FILES LICENSE DESTINATION share/${PROJECT_NAME})
install(DIRECTORY THIRD_PARTY_LICENSES DESTINATION share/${PROJECT_NAME})
ament_export_targets(export_controller_manager HAS_LIBRARY_TARGET)
ament_export_dependencies(Eigen3 safety_filter)
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_controller_manager test/test_controller_manager.cpp)
  target_link_libraries(test_controller_manager controller_manager)
  ament_add_gtest(test_homing_controller test/test_homing_controller.cpp)
  target_link_libraries(test_homing_controller controller_manager)
  ament_add_gtest(test_tracking test/test_tracking.cpp)
  target_link_libraries(test_tracking controller_manager)
  ament_target_dependencies(test_tracking quest_teleop_interfaces)
  find_package(ament_cmake_pytest REQUIRED)
  ament_add_pytest_test(test_controller_mock test/test_controller_mock.py TIMEOUT 60)
  ament_add_pytest_test(test_homing_mock test/test_homing_mock.py TIMEOUT 60)
  ament_add_pytest_test(test_cbf_runtime test/test_cbf_runtime.py TIMEOUT 60)
  find_package(rcl REQUIRED)
  add_library(slow_telemetry SHARED test/slow_telemetry.cpp)
  ament_target_dependencies(slow_telemetry rcl)
  target_link_libraries(slow_telemetry dl)
  ament_add_pytest_test(test_runtime_isolation test/test_runtime_isolation.py TIMEOUT 45
    ENV TELEOP_TEST_SLOW_PUBLISH_LIBRARY=${CMAKE_CURRENT_BINARY_DIR}/libslow_telemetry.so)
endif()
ament_package()
