cmake_minimum_required(VERSION 3.8)
project(safety_filter)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
find_package(ament_cmake REQUIRED)
find_package(Eigen3 REQUIRED)
# Pinned source dependency, fetched using third_party.repos. No host packages
# or Docker image changes. Keep OSQP out of ROS's signal-handler path.
set(PRINTING OFF CACHE BOOL "OSQP printing" FORCE)
set(CTRLC OFF CACHE BOOL "OSQP signal handlers" FORCE)
set(ENABLE_MKL_PARDISO OFF CACHE BOOL "OSQP MKL" FORCE)
set(UNITTESTS OFF CACHE BOOL "OSQP upstream tests" FORCE)
set(DFLOAT OFF CACHE BOOL "OSQP float precision" FORCE)
set(DLONG ON CACHE BOOL "OSQP index precision" FORCE)
set(OSQP_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../third_party/osqp_0_6_3")
if(NOT EXISTS "${OSQP_SOURCE_DIR}/CMakeLists.txt")
  message(FATAL_ERROR "Import pinned OSQP and its submodules; see docs/IMPLEMENTATION_PROGRESS.md")
endif()
add_subdirectory("${OSQP_SOURCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/osqp" EXCLUDE_FROM_ALL)
add_library(cbf_safety_filter src/cbf_safety_filter.cpp src/safety_coordinator.cpp)
target_include_directories(cbf_safety_filter PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<INSTALL_INTERFACE:include>)
target_link_libraries(cbf_safety_filter PUBLIC Eigen3::Eigen osqpstatic)


install(TARGETS cbf_safety_filter osqpstatic EXPORT export_safety_filter
  ARCHIVE DESTINATION lib LIBRARY DESTINATION lib RUNTIME DESTINATION bin)
install(DIRECTORY include/ DESTINATION include)
install(DIRECTORY "${OSQP_SOURCE_DIR}/include/" DESTINATION include/osqp FILES_MATCHING PATTERN "*.h")
install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME})
install(FILES LICENSE DESTINATION share/${PROJECT_NAME})
install(DIRECTORY THIRD_PARTY_LICENSES DESTINATION share/${PROJECT_NAME})
ament_export_targets(export_safety_filter HAS_LIBRARY_TARGET)
ament_export_dependencies(Eigen3)
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_cbf_safety_filter test/test_cbf_safety_filter.cpp)
  target_link_libraries(test_cbf_safety_filter cbf_safety_filter)
  target_link_options(test_cbf_safety_filter PRIVATE -Wl,--wrap=malloc -Wl,--wrap=calloc -Wl,--wrap=realloc)
  ament_add_gtest(test_safety_coordinator test/test_safety_coordinator.cpp)
  target_link_libraries(test_safety_coordinator cbf_safety_filter)
endif()
ament_package()
