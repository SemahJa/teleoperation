cmake_minimum_required(VERSION 3.8)
project(recording)
find_package(ament_cmake REQUIRED)
find_package(ament_cmake_python REQUIRED)
ament_python_install_package(${PROJECT_NAME})
install(PROGRAMS recording/menu_recorder.py DESTINATION lib/${PROJECT_NAME} RENAME menu_recorder)
install(PROGRAMS recording/export_tracking.py DESTINATION lib/${PROJECT_NAME} RENAME export_tracking)
install(PROGRAMS recording/validate_episode.py DESTINATION lib/${PROJECT_NAME} RENAME validate_episode)
install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_pytest REQUIRED)
  ament_add_pytest_test(test_demonstration test/test_demonstration.py)
  ament_add_pytest_test(test_menu_gate test/test_menu_gate.py)
  ament_add_pytest_test(test_configuration test/test_configuration.py)
  ament_add_pytest_test(test_export_tracking test/test_export_tracking.py)
  ament_add_pytest_test(test_clutch_filter test/test_clutch_filter.py)
  ament_add_pytest_test(test_export_lerobot test/test_export_lerobot.py)
  ament_add_pytest_test(test_supervision test/test_supervision.py)
  ament_add_pytest_test(test_reprocess_all test/test_reprocess_all.py)
endif()
ament_package()
