cmake_minimum_required(VERSION 3.8)
project(teleop_gripper)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
find_package(ament_cmake REQUIRED)
find_package(Franka REQUIRED)
find_package(rclcpp REQUIRED)
find_package(quest_teleop_interfaces REQUIRED)
add_library(hand_core src/hand_core.cpp)
target_include_directories(hand_core PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
add_executable(franka_hand_worker src/franka_hand_worker.cpp)
ament_target_dependencies(franka_hand_worker Franka rclcpp quest_teleop_interfaces)
target_link_libraries(franka_hand_worker hand_core)
install(TARGETS franka_hand_worker DESTINATION lib/${PROJECT_NAME})
install(TARGETS hand_core EXPORT export_hand_core ARCHIVE DESTINATION lib LIBRARY DESTINATION lib)
install(DIRECTORY include/ DESTINATION include)
install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME})
install(FILES LICENSE DESTINATION share/${PROJECT_NAME})
ament_export_targets(export_hand_core HAS_LIBRARY_TARGET)
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_hand_core test/test_hand_core.cpp)
  target_link_libraries(test_hand_core hand_core)
  find_package(ament_cmake_pytest REQUIRED)
  ament_add_pytest_test(test_hand_worker test/test_hand_worker.py TIMEOUT 60)
endif()
ament_package()
