cmake_minimum_required(VERSION 3.8)
project(panda_state_bringup)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
# Reuse the installation verified against this Panda; do not replace the SDK.
find_package(Franka 0.9.2 EXACT REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(nlohmann_json REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(std_msgs REQUIRED)
add_executable(read_only_state src/read_only_state.cpp)
target_include_directories(read_only_state PRIVATE include)
ament_target_dependencies(read_only_state rclcpp sensor_msgs geometry_msgs std_msgs)
target_link_libraries(read_only_state Franka::Franka Eigen3::Eigen nlohmann_json::nlohmann_json)
install(TARGETS read_only_state DESTINATION lib/${PROJECT_NAME})
install(PROGRAMS scripts/inspect_read_only_episode.py DESTINATION lib/${PROJECT_NAME} RENAME inspect_read_only_episode)
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  find_package(ament_cmake_pytest REQUIRED)
  ament_add_pytest_test(test_read_only_guards test/test_read_only_guards.py TIMEOUT 30 ENV READ_ONLY_TEST_BINARY=$<TARGET_FILE:read_only_state>)
  ament_add_gtest(test_snapshot test/test_snapshot.cpp)
  target_include_directories(test_snapshot PRIVATE include)
  ament_target_dependencies(test_snapshot rclcpp sensor_msgs geometry_msgs std_msgs)
  target_link_libraries(test_snapshot Franka::Franka Eigen3::Eigen nlohmann_json::nlohmann_json)
endif()
ament_package()
