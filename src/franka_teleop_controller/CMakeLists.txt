cmake_minimum_required(VERSION 3.8)
project(franka_teleop_controller)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()

find_package(ament_cmake REQUIRED)
find_package(Franka REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(franka_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(quest_teleop_interfaces REQUIRED)

# Pinned source dependency, fetched using third_party.repos. No host packages
# or Docker image changes. Keep OSQP out of ROS's signal-handler path.
set(PRINTING OFF CACHE BOOL "OSQP printing" FORCE)
set(CTRLC OFF CACHE BOOL "OSQP signal handlers" FORCE)
set(ENABLE_MKL_PARDISO OFF CACHE BOOL "OSQP MKL" FORCE)
set(UNITTESTS OFF CACHE BOOL "OSQP upstream tests" FORCE)
set(DFLOAT OFF CACHE BOOL "OSQP float precision" FORCE)
set(DLONG ON CACHE BOOL "OSQP index precision" FORCE)
set(OSQP_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../third_party/osqp_0_6_3")
if(NOT EXISTS "${OSQP_SOURCE_DIR}/CMakeLists.txt")
  message(FATAL_ERROR "Import pinned OSQP and its submodules; see docs/IMPLEMENTATION_PROGRESS.md")
endif()
add_subdirectory("${OSQP_SOURCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/osqp" EXCLUDE_FROM_ALL)
add_library(cbf_safety_filter src/cbf_safety_filter.cpp src/safety_coordinator.cpp)
target_include_directories(cbf_safety_filter PUBLIC include)
target_link_libraries(cbf_safety_filter PUBLIC Eigen3::Eigen osqpstatic)

add_library(control_core src/control_core.cpp src/hand_core.cpp src/homing_core.cpp src/joint_motion_generator.cpp)
target_include_directories(control_core PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<INSTALL_INTERFACE:include>)
target_link_libraries(control_core PUBLIC Eigen3::Eigen)
add_executable(cartesian_impedance_controller src/cartesian_impedance_controller.cpp)
ament_target_dependencies(cartesian_impedance_controller
  Franka franka_msgs geometry_msgs rclcpp sensor_msgs quest_teleop_interfaces)
target_link_libraries(cartesian_impedance_controller control_core cbf_safety_filter)
add_executable(franka_hand_worker src/franka_hand_worker.cpp)
ament_target_dependencies(franka_hand_worker Franka rclcpp quest_teleop_interfaces)
target_link_libraries(franka_hand_worker control_core)

install(TARGETS cartesian_impedance_controller franka_hand_worker DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME})
install(FILES LICENSE DESTINATION share/${PROJECT_NAME})
install(DIRECTORY THIRD_PARTY_LICENSES DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_control_core test/test_control_core.cpp)
  target_link_libraries(test_control_core control_core)
  ament_add_gtest(test_hand_core test/test_hand_core.cpp)
  target_link_libraries(test_hand_core control_core)
  ament_add_gtest(test_homing_core test/test_homing_core.cpp)
  target_link_libraries(test_homing_core control_core)
  ament_add_gtest(test_cbf_safety_filter test/test_cbf_safety_filter.cpp)
  target_link_libraries(test_cbf_safety_filter cbf_safety_filter)
  target_link_options(test_cbf_safety_filter PRIVATE
    -Wl,--wrap=malloc -Wl,--wrap=calloc -Wl,--wrap=realloc)
  ament_add_gtest(test_safety_coordinator test/test_safety_coordinator.cpp)
  target_link_libraries(test_safety_coordinator cbf_safety_filter)
  find_package(ament_cmake_pytest REQUIRED)
  ament_add_pytest_test(test_controller_mock test/test_controller_mock.py TIMEOUT 60)
  ament_add_pytest_test(test_hand_worker test/test_hand_worker.py TIMEOUT 60)
  ament_add_pytest_test(test_homing_mock test/test_homing_mock.py TIMEOUT 60)
  ament_add_pytest_test(test_cbf_runtime test/test_cbf_runtime.py TIMEOUT 60)
  ament_add_pytest_test(test_teleoperation_launch test/test_teleoperation_launch.py TIMEOUT 90)
endif()

ament_package()
